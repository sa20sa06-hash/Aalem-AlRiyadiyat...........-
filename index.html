decimal arithmetic parser. Supports + - * / and parentheses only. */ private class DecimalParser(private val source: String) { private val s = source.replace('×', '*').replace('÷', '/').replace('−', '-').filterNot { it.isWhitespace() } private var i = 0 fun parse(): BigDecimal { require(s.isNotEmpty()) { "أدخل تعبيراً" } val v = expression() require(i == s.length) { "رمز غير معروف عند الموضع ${i + 1}" } return v } private fun expression(): BigDecimal { var v = term() while (i < s.length && (s[i] == '+' || s[i] == '-')) { val op = s[i++]; val r = term(); v = if (op == '+') v.add(r) else v.subtract(r) } return v } private fun term(): BigDecimal { var v = factor() while (i < s.length && (s[i] == '*' || s[i] == '/')) { val op = s[i++]; val r = factor() v = if (op == '*') v.multiply(r) else v.divide(r, MathContext.DECIMAL128) } return v } private fun factor(): BigDecimal { if (i < s.length && (s[i] == '+' || s[i] == '-')) { val neg = s[i++] == '-'; val v = factor(); return if (neg) v.negate() else v } if (i < s.length && s[i] == '(') { i++; val v = expression(); require(i < s.length && s[i] == ')') { "قوس إغلاق مفقود" }; i++; return v } val start = i while (i < s.length && (s[i].isDigit() || s[i] == '.')) i++ require(i > start) { "عدد متوقع" } return BigDecimal(s.substring(start, i)) } }
